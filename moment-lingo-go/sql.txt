create table `user`
(
    user_id    bigint primary key,
    nickname   varchar(255)                                                    not null,
    avatar_url varchar(255)                                                    not null,
    passcode   varchar(255),
    email      varchar(255) unique,
    phone      varchar(255) unique,
    created_at timestamp default current_timestamp                             not null,
    updated_at timestamp default current_timestamp on update current_timestamp not null
);

create table vocabulary
(
    vocabulary_id int primary key auto_increment,
    vocabulary    varchar(255) not null unique,
    phonetic      varchar(255),
    translation   json,
    explanation   json,
    tags          json,
    did           varchar(255),
    done          varchar(255),
    doing         varchar(255),
    does          varchar(255),
    comparative   varchar(255),
    superlative   varchar(255),
    plural        varchar(255),
    lemma         varchar(255),
    lexical_id    int,
    `type`        varchar(255)
) collate utf8mb4_bin;


create table word
(
    word_id       int primary key auto_increment,
    vocabulary_id int          not null,
    word          varchar(255) not null unique,
    brief         json,
    content       json,
    tags          json,
    translation   json,
    foreign key (vocabulary_id) references vocabulary (vocabulary_id)
) collate utf8mb4_bin;


create table phrase
(
    phrase_id     int primary key auto_increment,
    vocabulary_id int          not null,
    word_id       int          not null references word (word_id),
    phrase        varchar(255) not null unique,
    lemma         varchar(255) not null,
    translation   json,
    tags          json,
    foreign key (vocabulary_id) references vocabulary (vocabulary_id)

) collate utf8mb4_bin;



create table voice
(
    voice_id   int primary key auto_increment,
    voice_name varchar(255)                                                    not null,
    voice_key  varchar(255)                                                    not null unique,
    `type`     varchar(255)                                                    not null,
    model      varchar(255)                                                    not null,
    tags       json                                                            not null,
    weight     int                                                             not null default 0,
    created_at timestamp default current_timestamp                             not null,
    updated_at timestamp default current_timestamp on update current_timestamp not null
);

create table essay
(
    essay_id   bigint primary key,
    user_id    bigint                                                          not null,
    content    text                                                            not null,
    result     json,
    `type`     int                                                             not null,
    created_at timestamp default current_timestamp                             not null,
    updated_at timestamp default current_timestamp on update current_timestamp not null,
    foreign key (user_id) references `user` (user_id)
);



create table book
(
    book_id        int auto_increment primary key,
    book_origin_id varchar(255),
    word_count     int                                                             not null,
    title          varchar(255)                                                    not null,
    tags           json                                                            not null,
    created_at     timestamp default current_timestamp                             not null,
    updated_at     timestamp default current_timestamp on update current_timestamp not null
);

create table book_vocabulary
(
    book_vocabulary_id int auto_increment primary key,
    book_id            int not null,
    vocabulary_id      int not null,
    `rank`             int not null,
    foreign key (book_id) references book (book_id),
    foreign key (vocabulary_id) references vocabulary (vocabulary_id)
);
